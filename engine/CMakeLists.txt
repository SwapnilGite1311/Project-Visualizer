cmake_minimum_required(VERSION 3.28)
project(pv_engine VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(PV_ENGINE_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
set(PV_ENGINE_VERSION ${PROJECT_VERSION})

option(PV_BUILD_TESTS "Build unit tests" ON)

include(FetchContent)
if(PV_BUILD_TESTS)
  enable_testing()
  include(GoogleTest)
endif()
include(${PV_ENGINE_ROOT}/cmake/PvModule.cmake)
include(${PV_ENGINE_ROOT}/cmake/GenerateRegistry.cmake)

if(MSVC)
  add_compile_options(/utf-8 /EHsc /permissive- /Zc:__cplusplus)
endif()

# Third-party libraries: every engine/third_party/<name>/CMakeLists.txt (alphabetical)
file(GLOB PV_TP_CMAKES CONFIGURE_DEPENDS ${PV_ENGINE_ROOT}/third_party/*/CMakeLists.txt)
list(SORT PV_TP_CMAKES)
foreach(cm ${PV_TP_CMAKES})
  get_filename_component(dir ${cm} DIRECTORY)
  add_subdirectory(${dir})
endforeach()

# Engine modules: every engine/modules/<name>/CMakeLists.txt (alphabetical)
file(GLOB PV_MODULE_CMAKES CONFIGURE_DEPENDS ${PV_ENGINE_ROOT}/modules/*/CMakeLists.txt)
list(SORT PV_MODULE_CMAKES)
foreach(cm ${PV_MODULE_CMAKES})
  get_filename_component(dir ${cm} DIRECTORY)
  add_subdirectory(${dir})
endforeach()

# The pv-engine executable (created by P1-EAPP)
if(EXISTS ${PV_ENGINE_ROOT}/app/CMakeLists.txt)
  add_subdirectory(${PV_ENGINE_ROOT}/app)
endif()
